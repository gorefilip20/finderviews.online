function mapGreenhouseRuntimeJob(job, board, now = Date.now()) { if (!job.title || !job.absolute_url || !job.content) return null; const publishedMs = Date.parse(job.updated_at || new Date().toISOString()); const ageMs = now - publishedMs; if (!Number.isFinite(publishedMs) || ageMs > MAX_JOB_AGE_DAYS * 24 * 60 * 60 * 1000 || ageMs < -12 * 60 * 60 * 1000) return null; const description = stripMarkup(job.content).slice(0, 7000); const location = job.location?.name || (job.offices || []).map((office) => office.name).filter(Boolean).join(", ") || "Not specified"; return { id: "greenhouse-" + board + "-" + (job.id || job.absolute_url), title: stripMarkup(job.title), company: board, geography: location, industry: [], jobType: ["Full-time"], level: "Not specified", excerpt: description.slice(0, 480), description, postedAt: new Date(publishedMs).toISOString(), ageHours: Math.max(0, Math.floor(ageMs / (60 * 60 * 1000))), sourceUrl: asSafeSourceUrl(job.absolute_url), sourceName: "Greenhouse", contactStatus: "Apply through the original employer board.", companyWebsite: undefined, applyEmail: undefined, contactSearchUrl: "https://www.google.com/search?q=" + encodeURIComponent(board + " careers"), hasActionableContact: Boolean(job.absolute_url) }; }
async function fetchGreenhouseRuntime() { const boards = (process.env.GREENHOUSE_BOARDS || "").split(",").map((value) => value.trim()).filter(Boolean); return (await Promise.all(boards.map(async (board) => { const response = await fetch("https://boards-api.greenhouse.io/v1/boards/" + encodeURIComponent(board) + "/jobs?content=true", { headers: { Accept: "application/json", "User-Agent": "Finderviews/1.0" }, signal: AbortSignal.timeout(12000) }); if (!response.ok) return []; const payload = await response.json(); return (payload.jobs || []).map((job) => mapGreenhouseRuntimeJob(job, board)).filter(Boolean); }))).flat(); }
function mapLeverRuntimeJob(job, site, now = Date.now()) { if (!job.text || !(job.hostedUrl || job.applyUrl) || !job.createdAt) return null; const publishedMs = Number(job.createdAt); const ageMs = now - publishedMs; if (!Number.isFinite(publishedMs) || ageMs > MAX_JOB_AGE_DAYS * 24 * 60 * 60 * 1000 || ageMs < -12 * 60 * 60 * 1000) return null; const description = stripMarkup(job.descriptionPlain || "").slice(0, 7000); return { id: "lever-" + site + "-" + (job.id || job.hostedUrl), title: stripMarkup(job.text), company: site, geography: stripMarkup(job.categories?.location) || "Not specified", industry: [stripMarkup(job.categories?.team)].filter(Boolean), jobType: [stripMarkup(job.categories?.commitment)].filter(Boolean), level: "Not specified", excerpt: description.slice(0, 480), description, postedAt: new Date(publishedMs).toISOString(), ageHours: Math.max(0, Math.floor(ageMs / (60 * 60 * 1000))), sourceUrl: asSafeSourceUrl(job.hostedUrl || job.applyUrl), sourceName: "Lever", contactStatus: "Apply through the original employer board.", companyWebsite: undefined, applyEmail: undefined, contactSearchUrl: "https://www.google.com/search?q=" + encodeURIComponent(site + " careers"), hasActionableContact: Boolean(job.hostedUrl || job.applyUrl) }; }
async function fetchLeverRuntime() { const sites = (process.env.LEVER_SITES || "").split(",").map((value) => value.trim()).filter(Boolean); return (await Promise.all(sites.map(async (site) => { const response = await fetch("https://api.lever.co/v0/postings/" + encodeURIComponent(site) + "?mode=json", { headers: { Accept: "application/json", "User-Agent": "Finderviews/1.0" }, signal: AbortSignal.timeout(12000) }); if (!response.ok) return []; const payload = await response.json(); return (payload || []).map((job) => mapLeverRuntimeJob(job, site)).filter(Boolean); }))).flat(); }
